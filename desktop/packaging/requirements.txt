# Sidecar build dependencies (personal edition)
#
# WARNING: this file must stay pure ASCII.
#   pip reads requirements files using the system locale encoding
#   (GBK on Chinese Windows) -> non-ASCII comments crash with
#   "UnicodeDecodeError: 'gbk' codec can't decode byte ...".
#   Chinese docs live in desktop/BUILD.md instead.
#
# These packages MUST be installed in the Python used for packaging:
#   PyInstaller only bundles modules that are actually importable.
#   Miss one grammar package and the built exe fails at runtime with
#   "no parser available" - while packaging itself reports success.
#
# Versions are kept identical to app/requirements.txt so both the
# community server and the desktop sidecar parse code the same way.

# packager
pyinstaller>=6.10,<7

# file filtering: gitignore semantics (MIT). Without it the sidecar falls
# back to a simplified matcher, so this is a quality upgrade, not a hard dep.
pathspec>=0.12,<1
